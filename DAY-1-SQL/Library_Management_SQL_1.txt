-- A1 – DDL

-- 1. Create Database
CREATE DATABASE LibraryDB;
USE LibraryDB;

-- 2. Create Books Table
CREATE TABLE Books(
book_id INT PRIMARY KEY,
book_name VARCHAR(100),
author VARCHAR(100),
price DECIMAL(10,2)
);

-- 3. Create Members Table
CREATE TABLE Members(
member_id INT PRIMARY KEY,
member_name VARCHAR(100),
city VARCHAR(50),
phone VARCHAR(15)
);

-- 4. Create Borrow Table
CREATE TABLE Borrow(
borrow_id INT PRIMARY KEY,
book_id INT,
member_id INT,
borrow_date DATE
);

-- 5. Add Category
ALTER TABLE Books ADD category VARCHAR(50);

-- 6. Add Quantity
ALTER TABLE Books ADD quantity INT;

-- 7. Add Email
ALTER TABLE Members ADD email VARCHAR(100);

-- 8. Add Return Date
ALTER TABLE Borrow ADD return_date DATE;

-- 9. Modify Price
ALTER TABLE Books MODIFY price DECIMAL(10,2);

-- 10. Rename Quantity
ALTER TABLE Books RENAME COLUMN quantity TO stock_quantity;

-- 11. Display Books Structure
DESC Books;

-- 12. Display Members Structure
DESC Members;

-- 13. Display Borrow Structure
DESC Borrow;

-- A4 – DCL

-- 38. Create User
CREATE USER 'library_user'@'localhost'
IDENTIFIED BY 'Library@123';

-- 39. Grant SELECT
GRANT SELECT ON LibraryDB.Books
TO 'library_user'@'localhost';

-- 40. Grant INSERT
GRANT INSERT ON LibraryDB.Books
TO 'library_user'@'localhost';

-- 41. Grant UPDATE
GRANT UPDATE ON LibraryDB.Books
TO 'library_user'@'localhost';

-- 42. Display Privileges
SHOW GRANTS FOR 'library_user'@'localhost';

-- 43. Revoke INSERT
REVOKE INSERT ON LibraryDB.Books
FROM 'library_user'@'localhost';

-- 44. Revoke UPDATE
REVOKE UPDATE ON LibraryDB.Books
FROM 'library_user'@'localhost';

-- 45. Grant SELECT on All Tables
GRANT SELECT ON LibraryDB.*
TO 'library_user'@'localhost';

-- 46. Revoke SELECT from Books
REVOKE SELECT ON LibraryDB.Books
FROM 'library_user'@'localhost';

-- 47. Display Final Privileges
SHOW GRANTS FOR 'library_user'@'localhost';
